# Dial Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Dial — a native macOS menubar HUD for Spotify that shows "what this song is to *you*" (play count, rank, badges) with a pixel-pet mascot that grows from an egg as you listen.

**Architecture:** Single-process SwiftUI + AppKit app. Local SQLite (GRDB) for plays/tracks/artists/pet_state. Live polling of Spotify Web API (every 3s) + import of Spotify's Extended Streaming History export. `PetEngine` awards XP in-transaction with each qualifying play. All UI reads from `@Observable` ViewModels fed by `AggregationsCache` and `PetEngine`.

**Tech Stack:** Swift 5.10+, SwiftUI (macOS 14), `NSStatusItem`/`NSPopover` or `MenuBarExtra`, GRDB.swift (SQLite), KeychainAccess, `URLSession`, `ASWebAuthenticationSession`, `swift-snapshot-testing` (tests), XCTest.

**Reference spec:** `docs/superpowers/specs/2026-04-24-dial-design.md`

```swift
// Dial/Store/Migrations.swift
import GRDB
import Foundation

enum Migrations {
  static func register() -> DatabaseMigrator {
    var m = DatabaseMigrator()
    m.registerMigration("v1_initial") { db in
      try db.create(table: "plays") { t in
        t.autoIncrementedPrimaryKey("id")
        t.column("played_at", .integer).notNull()
        t.column("track_uri", .text).notNull()
        t.column("ms_played", .integer).notNull()
        t.column("source",    .text).notNull()
        t.uniqueKey(["played_at", "track_uri", "ms_played"])
      }
      try db.create(index: "plays_by_track", on: "plays", columns: ["track_uri", "played_at"])
      try db.create(index: "plays_by_time",  on: "plays", columns: ["played_at"])

      try db.create(table: "tracks") { t in
        t.column("uri", .text).primaryKey()
        t.column("name", .text)
        t.column("album_name", .text)
        t.column("album_art_url", .text)
        t.column("duration_ms", .integer)
        t.column("artists_json", .text)
      }
      try db.create(table: "artists") { t in
        t.column("uri", .text).primaryKey()
        t.column("name", .text)
        t.column("genres_json", .text)
        t.column("image_url", .text)
      }
      try db.create(table: "imports_log") { t in
        t.autoIncrementedPrimaryKey("id")
        t.column("imported_at", .integer).notNull()
        t.column("source_path", .text)
        t.column("rows_inserted", .integer)
        t.column("notes", .text)
      }
      try db.create(table: "pet_state") { t in
        t.column("id", .integer).primaryKey().check { $0 == 1 }
        t.column("xp", .integer).notNull().defaults(to: 0)
        t.column("stage", .text).notNull().defaults(to: "egg")
        t.column("hatched_at", .integer)
        t.column("last_play_at", .integer)
        t.column("streak_days", .integer).notNull().defaults(to: 0)
        t.column("streak_anchor_date", .text)
        t.column("elder_unlocked_at", .integer)
        t.column("installed_at", .integer).notNull()
      }
      try db.execute(sql: """
        INSERT INTO pet_state (id, xp, stage, streak_days, installed_at)
```

```swift
// Dial/Store/DatabaseStack.swift
import GRDB
import Foundation

final class DatabaseStack {
  let dbq: DatabaseQueue

  init(url: URL) throws {
    var config = Configuration()
    config.foreignKeysEnabled = true
    dbq = try DatabaseQueue(path: url.path, configuration: config)
    try Migrations.register().migrate(dbq)
  }

  static func defaultURL() throws -> URL {
    let base = FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
      .appendingPathComponent("Dial", isDirectory: true)
    try FileManager.default.createDirectory(at: base, withIntermediateDirectories: true)
    return base.appendingPathComponent("dial.sqlite")
  }
}
```

```swift
// Dial/Store/DAO/PlaysDAO.swift
import GRDB

struct PlaysDAO {
  let dbq: DatabaseQueue

  func insert(_ play: Play) throws {
    var p = play
    try dbq.write { db in try p.insert(db) }
  }
  func playsFor(trackUri: String) throws -> [Play] {
    try dbq.read { db in
      try Play.filter(Play.Columns.trackUri == trackUri)
        .order(Play.Columns.playedAt.desc)
        .fetchAll(db)
    }
  }
  func recent(limit: Int) throws -> [Play] {
    try dbq.read { db in
      try Play.order(Play.Columns.playedAt.desc).limit(limit).fetchAll(db)
    }
  }
}
```

```swift
import GRDB

struct TracksDAO {
  let dbq: DatabaseQueue
  func upsert(_ t: Track) throws { try dbq.write { db in try t.save(db) } }
  func fetch(uri: String) throws -> Track? {
    try dbq.read { db in try Track.fetchOne(db, key: uri) }
  }
  func missingUris(among uris: [String]) throws -> [String] {
    guard !uris.isEmpty else { return [] }
    return try dbq.read { db in
      let present = try String.fetchAll(db,
        sql: "SELECT uri FROM tracks WHERE uri IN (\(uris.map{_ in "?"}.joined(separator: ",")))",
        arguments: StatementArguments(uris))
      let set = Set(present)
      return uris.filter { !set.contains($0) }
    }
  }
}
```

```swift
import GRDB

struct PetStateDAO {
  let dbq: DatabaseQueue
  func get() throws -> PetState {
    try dbq.read { db in
      guard let s = try PetState.fetchOne(db, key: 1) else {
        throw NSError(domain: "Dial.Store", code: 1, userInfo: [NSLocalizedDescriptionKey: "missing pet_state singleton"])
      }
      return s
    }
  }
  func update(_ mutate: (inout PetState) -> Void) throws {
    try dbq.write { db in
      var s = try PetState.fetchOne(db, key: 1)!
      mutate(&s)
      try s.update(db)
    }
  }
  /// For use inside a larger transaction (called by `PetEngine` from within the same write).
  func update(in db: Database, _ mutate: (inout PetState) -> Void) throws {
    var s = try PetState.fetchOne(db, key: 1)!
    mutate(&s)
    try s.update(db)
  }
}
```

```swift
// Dial/Aggregations/PlayAggregations.swift
import GRDB
import Foundation

struct PlayAggregations {
  let dbq: DatabaseQueue
  var now: () -> Int64 = { Int64(Date().timeIntervalSince1970) }

  func plays30d(trackUri: String) throws -> Int {
    let cutoff = now() - 30 * 86_400
    return try dbq.read { db in
      try Int.fetchOne(db,
        sql: "SELECT COUNT(*) FROM plays WHERE track_uri = ? AND played_at > ?",
        arguments: [trackUri, cutoff]) ?? 0
    }
  }
  /// Returns 1-based rank within last 7d among all tracks played in that window.
  /// Ties broken by most-recent play.
  func rank7d(trackUri: String) throws -> Int? {
    let cutoff = now() - 7 * 86_400
    return try dbq.read { db in
      try Int.fetchOne(db, sql: """
        WITH ranked AS (
          SELECT track_uri, COUNT(*) as c, MAX(played_at) as last
          FROM plays WHERE played_at > ? GROUP BY track_uri
        )
        SELECT rnk FROM (
          SELECT track_uri, ROW_NUMBER() OVER (ORDER BY c DESC, last DESC) as rnk FROM ranked
        ) WHERE track_uri = ?
        """, arguments: [cutoff, trackUri])
    }
  }
}
```

```swift
  func firstListen(trackUri: String) throws -> Bool {
```

```swift
extension PlayAggregations {
  struct SessionPlay { let playedAt: Int64; let trackUri: String; let msPlayed: Int64 }

  func last60MinTracks() throws -> [SessionPlay] {
    let cutoff = now() - 3600
    return try dbq.read { db in
      try db.fetchAll(sql: """
        SELECT played_at, track_uri, ms_played FROM plays
        WHERE played_at > ? ORDER BY played_at DESC
      """, arguments: [cutoff]).map { row in
        SessionPlay(playedAt: row["played_at"], trackUri: row["track_uri"], msPlayed: row["ms_played"])
      }
    }
  }

  /// Derives primary-artist URIs from `tracks.artists_json[0].uri`.
  /// Ranks artists by unique-day-weighted play count in the last 7d, ties broken by recent play.
  func artistRank7d(artistUri: String) throws -> Int? {
    let cutoff = now() - 7 * 86_400
    return try dbq.read { db in

          SELECT json_extract(t.artists_json, '$[0].uri') AS aurl, p.played_at
          FROM plays p JOIN tracks t ON t.uri = p.track_uri
          WHERE p.played_at > ? AND t.artists_json IS NOT NULL
        ),
        ranked AS (
          SELECT aurl, COUNT(*) AS c, MAX(played_at) AS last
          FROM artist_plays GROUP BY aurl
        )
        SELECT rnk FROM (
          SELECT aurl, ROW_NUMBER() OVER (ORDER BY c DESC, last DESC) AS rnk FROM ranked
        ) WHERE aurl = ?
      """, arguments: [cutoff, artistUri])
    }
```

```swift
// Dial/Auth/PKCE.swift
import Foundation
import CryptoKit

enum PKCE {
  static func makeCodeVerifier(length: Int = 64) -> String {
    let chars = Array("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-._~")
    var rng = SystemRandomNumberGenerator()
    return String((0..<length).map { _ in chars[Int(rng.next(upperBound: UInt64(chars.count)))] })
  }
  static func makeCodeChallenge(for verifier: String) -> String {
    let hash = SHA256.hash(data: Data(verifier.utf8))
    return Data(hash).base64URLEncoded()
  }
}
private extension Data {
  func base64URLEncoded() -> String {
    base64EncodedString()
      .replacingOccurrences(of: "+", with: "-")
      .replacingOccurrences(of: "/", with: "_")
      .replacingOccurrences(of: "=", with: "")
  }
}
```

```swift
// Dial/Auth/SpotifyAuthService.swift
import Foundation
import AuthenticationServices

final class SpotifyAuthService: NSObject {
  private let clientID: String
  private let redirectURI = "dial://callback"
  private let scopes = [
    "user-read-currently-playing","user-read-playback-state",
    "user-modify-playback-state","user-read-recently-played"
  ]
  private let tokenStore: TokenStore

  init(clientID: String, tokenStore: TokenStore) {
    self.clientID = clientID
    self.tokenStore = tokenStore
  }

  @MainActor func signIn() async throws -> SpotifyToken {
    let verifier = PKCE.makeCodeVerifier()
    let challenge = PKCE.makeCodeChallenge(for: verifier)

    var auth = URLComponents(string: "https://accounts.spotify.com/authorize")!
    auth.queryItems = [
      .init(name: "client_id", value: clientID),
      .init(name: "response_type", value: "code"),
      .init(name: "redirect_uri", value: redirectURI),
      .init(name: "code_challenge_method", value: "S256"),
      .init(name: "code_challenge", value: challenge),
      .init(name: "scope", value: scopes.joined(separator: " "))
    ]

    let code: String = try await withCheckedThrowingContinuation { cont in
      let session = ASWebAuthenticationSession(url: auth.url!, callbackURLScheme: "dial") { callback, error in
        if let error { cont.resume(throwing: error); return }
        guard let cb = callback,
              let c = URLComponents(url: cb, resolvingAgainstBaseURL: false)?.queryItems?.first(where: { $0.name == "code" })?.value
        else { cont.resume(throwing: NSError(domain: "Dial.Auth", code: 1)); return }
        cont.resume(returning: c)
      }
      session.presentationContextProvider = self
      session.prefersEphemeralWebBrowserSession = false
      session.start()
    }

    let token = try await exchange(code: code, verifier: verifier)
    try tokenStore.save(token)
    return token
  }

  private func exchange(code: String, verifier: String) async throws -> SpotifyToken {
    var req = URLRequest(url: URL(string: "https://accounts.spotify.com/api/token")!)
    req.httpMethod = "POST"
    req.setValue("application/x-www-form-urlencoded", forHTTPHeaderField: "Content-Type")
    let body = [
      "grant_type=authorization_code",
      "code=\(code)",
      "redirect_uri=\(redirectURI)",
      "client_id=\(clientID)",
      "code_verifier=\(verifier)"
    ].joined(separator: "&")
    req.httpBody = body.data(using: .utf8)
    let (data, _) = try await URLSession.shared.data(for: req)
    struct R: Decodable { let access_token: String; let refresh_token: String; let expires_in: Int }
    let r = try JSONDecoder().decode(R.self, from: data)
    return .init(accessToken: r.access_token, refreshToken: r.refresh_token,
                 expiresAt: Int64(Date().timeIntervalSince1970) + Int64(r.expires_in))
  }

  func refresh() async throws -> SpotifyToken {
    guard let current = try tokenStore.load() else {
      throw NSError(domain: "Dial.Auth", code: 2, userInfo: [NSLocalizedDescriptionKey: "no token"])
    }
    var req = URLRequest(url: URL(string: "https://accounts.spotify.com/api/token")!)
    req.httpMethod = "POST"
    req.setValue("application/x-www-form-urlencoded", forHTTPHeaderField: "Content-Type")
    let body = "grant_type=refresh_token&refresh_token=\(current.refreshToken)&client_id=\(clientID)"
    req.httpBody = body.data(using: .utf8)
    let (data, _) = try await URLSession.shared.data(for: req)
    struct R: Decodable { let access_token: String; let refresh_token: String?; let expires_in: Int }
    let r = try JSONDecoder().decode(R.self, from: data)
    let updated = SpotifyToken(
      accessToken: r.access_token,
      refreshToken: r.refresh_token ?? current.refreshToken,
      expiresAt: Int64(Date().timeIntervalSince1970) + Int64(r.expires_in)
    )
    try tokenStore.save(updated)
    return updated
  }
}

extension SpotifyAuthService: ASWebAuthenticationPresentationContextProviding {
  func presentationAnchor(for session: ASWebAuthenticationSession) -> ASPresentationAnchor {
    NSApplication.shared.windows.first ?? ASPresentationAnchor()
  }
}
```

```swift
// Dial/Spotify/SpotifyClient.swift
import Foundation

final class SpotifyClient {
  private let session: URLSession
  private let tokenStore: TokenStore
  private let refresh: () async throws -> SpotifyToken

  init(session: URLSession = .shared, tokenStore: TokenStore,
       refresh: @escaping () async throws -> SpotifyToken) {
    self.session = session
    self.tokenStore = tokenStore
    self.refresh = refresh
  }

  private func authedGET(_ url: URL, retryOn401: Bool = true) async throws -> (Data, HTTPURLResponse) {
    var token = try tokenStore.load()
    var req = URLRequest(url: url)
    req.setValue("Bearer \(token?.accessToken ?? "")", forHTTPHeaderField: "Authorization")
    let (data, resp) = try await session.data(for: req)
    let http = resp as! HTTPURLResponse
    if http.statusCode == 401, retryOn401 {
      token = try await refresh()
      return try await authedGET(url, retryOn401: false)
    }
    return (data, http)
  }

  func currentlyPlaying() async throws -> CurrentlyPlaying? {
    let url = URL(string: "https://api.spotify.com/v1/me/player/currently-playing")!
    let (data, http) = try await authedGET(url)
    if http.statusCode == 204 { return nil }
    return try JSONDecoder().decode(CurrentlyPlaying.self, from: data)
  }

  func audioFeatures(trackId: String) async throws -> AudioFeatures {
    let url = URL(string: "https://api.spotify.com/v1/audio-features/\(trackId)")!
    let (data, _) = try await authedGET(url)
    return try JSONDecoder().decode(AudioFeatures.self, from: data)
  }
}
```

```swift
// Dial/Polling/PollingService.swift
import Foundation

protocol PollingClient { func currentlyPlaying() async throws -> CurrentlyPlaying? }
extension SpotifyClient: PollingClient {}

final class PollingService {
  private let client: PollingClient
  private let interval: TimeInterval
  private let clock: () -> Int64
  private var lastTrackUri: String?
  private var counted = false
  private var playbackState: Bool? = nil     // nil = unknown; true = playing; false = paused

  init(client: PollingClient, interval: TimeInterval = 3.0, clock: @escaping () -> Int64 = { Int64(Date().timeIntervalSince1970) }) {
    self.client = client
    self.interval = interval
    self.clock = clock
  }

  func stream() -> AsyncThrowingStream<PlaybackEvent, Error> {
    AsyncThrowingStream { continuation in
      Task {
        while !Task.isCancelled {
          do {
            let cp = try await client.currentlyPlaying()
            for e in interpret(cp) { continuation.yield(e) }
          } catch {
            continuation.finish(throwing: error); return
          }
          if interval > 0 { try? await Task.sleep(nanoseconds: UInt64(interval * 1_000_000_000)) }
        }
        continuation.finish()
      }
    }
  }

  private func interpret(_ cp: CurrentlyPlaying?) -> [PlaybackEvent] {
    guard let cp, let item = cp.item else {
      lastTrackUri = nil; counted = false; playbackState = nil
      return [.noPlayback]
    }
    var out: [PlaybackEvent] = []
    if item.uri != lastTrackUri {
      lastTrackUri = item.uri; counted = false
      out.append(.started(
        trackUri: item.uri, name: item.name,
        artistName: item.artists.first?.name ?? "",
        artistUri: item.artists.first?.uri,
        albumName: item.album.name,
        artUrl: item.album.images.first?.url,
        durationMs: item.duration_ms,
        at: clock()))
    }
    if let p = cp.progress_ms {
      out.append(.progress(ms: p))
      if !counted, DedupRule.shouldCount(progressMs: p, durationMs: item.duration_ms) {
        counted = true
        out.append(.countedPlay(trackUri: item.uri, msPlayed: Int64(p), at: clock()))
      }
    }
    if playbackState != cp.is_playing {
      out.append(cp.is_playing ? .resumed : .paused)
      playbackState = cp.is_playing
    }
    return out
  }
}
```

```swift
import Foundation

enum VarietyBonus { case none, newAlbum, newArtist }

enum XPFormula {
  static func compute(msPlayed: Int64, variety: VarietyBonus, streakDays: Int64) -> Int64 {
    let base = Double(msPlayed) / 60_000.0

    let mult: Double = streakDays < 3 ? 1.0 : min(1 + Double(streakDays - 2) * 0.25, 2.0)

  }
```

```swift
// Dial/Pet/StreakCalendar.swift
import Foundation

struct StreakCalendar {
  struct Result { let anchor: String; let days: Int64 }
  func apply(playDate: String, priorAnchor: String?, priorDays: Int64) -> Result {
    guard let priorAnchor else { return .init(anchor: playDate, days: 1) }
    if playDate == priorAnchor { return .init(anchor: priorAnchor, days: priorDays) }
    guard let prior = date(from: priorAnchor), let today = date(from: playDate) else {
      return .init(anchor: playDate, days: 1)
    }
    let gap = Calendar(identifier: .gregorian).dateComponents([.day], from: prior, to: today).day ?? 0
    return gap == 1 ? .init(anchor: playDate, days: priorDays + 1)
                    : .init(anchor: playDate, days: 1)
  }
  private func date(from ymd: String) -> Date? {
    let f = DateFormatter(); f.dateFormat = "yyyy-MM-dd"; f.timeZone = TimeZone.current
    return f.date(from: ymd)
  }
  static func ymd(from unixSeconds: Int64, timeZone: TimeZone = .current) -> String {
    let f = DateFormatter(); f.dateFormat = "yyyy-MM-dd"; f.timeZone = timeZone
    return f.string(from: Date(timeIntervalSince1970: TimeInterval(unixSeconds)))
  }
}
```

```swift
// Dial/Pet/PetEngine.swift
import GRDB
import Foundation

final class PetEngine {
  private let dbq: DatabaseQueue
  private let clock: () -> Int64
  private let elderLockDays: Int64 = 30

  init(dbq: DatabaseQueue, clock: @escaping () -> Int64 = { Int64(Date().timeIntervalSince1970) }) {
    self.dbq = dbq; self.clock = clock
  }

  func recordPlay(_ play: Play) throws {
    try dbq.write { db in
      var p = play
      try p.insert(db)

      var state = try PetState.fetchOne(db, key: 1)!

      let variety = try Self.variety(for: play, in: db)
      let day = StreakCalendar.ymd(from: play.playedAt)
      let streak = StreakCalendar().apply(playDate: day, priorAnchor: state.streakAnchorDate, priorDays: state.streakDays)

      let xpDelta = XPFormula.compute(msPlayed: play.msPlayed, variety: variety, streakDays: streak.days)

      // Cap imported XP at companion threshold.
      let capCompanion: Int64 = 25_000
      let proposed = state.xp + xpDelta
      if play.source == "import" {
        state.xp = min(proposed, capCompanion)
      } else {
        state.xp = proposed
      }

      state.streakAnchorDate = streak.anchor
      state.streakDays = streak.days
      state.lastPlayAt = play.playedAt

      // Elder unlock: 30 calendar days since install.
      let elderUnlocked = (clock() - state.installedAt) >= elderLockDays * 86_400
      if elderUnlocked, state.elderUnlockedAt == nil { state.elderUnlockedAt = clock() }

      let newStage = PetStageTable.stage(forXP: state.xp, elderUnlocked: elderUnlocked)
      if newStage == .hatching, state.hatchedAt == nil { state.hatchedAt = clock() }
      state.stage = newStage

      try state.update(db)
    }
  }

  private static func variety(for play: Play, in db: Database) throws -> VarietyBonus {
    // Look up the track's primary artist & album, then check prior plays.
    let track = try Track.fetchOne(db, key: play.trackUri)
    let artistUri: String? = {
      guard let s = track?.artistsJson?.data(using: .utf8) else { return nil }
      struct A: Decodable { let uri: String }
      return (try? JSONDecoder().decode([A].self, from: s))?.first?.uri
    }()
    // newArtist: no other plays of tracks by this artist ever
    if let au = artistUri {
      let count = try Int.fetchOne(db, sql: """
        SELECT COUNT(*) FROM plays p JOIN tracks t ON t.uri = p.track_uri
        WHERE p.id != ? AND json_extract(t.artists_json, '$[0].uri') = ?
      """, arguments: [play.id ?? -1, au]) ?? 0
      if count == 0 { return .newArtist }
    }
    // newAlbum: no other plays of tracks from this album
    if let albumName = track?.albumName, let artistUri {
      let count = try Int.fetchOne(db, sql: """
        SELECT COUNT(*) FROM plays p JOIN tracks t ON t.uri = p.track_uri
        WHERE p.id != ? AND t.album_name = ? AND json_extract(t.artists_json, '$[0].uri') = ?
      """, arguments: [play.id ?? -1, albumName, artistUri]) ?? 0
      if count == 0 { return .newAlbum }
    }
    return .none
  }
}
```

```swift
// Dial/Pet/PetBehavior.swift
import Foundation

enum PetBehavior: Equatable {
  case idle, beatSync(bpm: Double), firstListen, rediscovered, paused, dormant, levelUp
}

struct PetContext {
  let playing: Bool
  let paused: Bool
  let firstListen: Bool
  let rediscovered: Bool
  let leveledUp: Bool
  let lastPlayAt: Int64?
  let now: Int64
  let bpm: Double?
}

enum PetBehaviorResolver {
  static func resolve(_ c: PetContext) -> PetBehavior {
    if c.leveledUp { return .levelUp }
    if c.firstListen { return .firstListen }
    if c.rediscovered { return .rediscovered }
    if c.paused { return .paused }
    if let lp = c.lastPlayAt, c.now - lp >= 7 * 86_400 { return .dormant }
    if c.playing { return c.bpm.map(PetBehavior.beatSync) ?? .idle }
    return .idle
  }
}
```

```swift
  /// `dots` are (x,y) single amber pixels.

    for (x, y) in dots where x >= 0 && x < 14 && y >= 0 && y < 14 { g[y][x] = 1 }
    for (x, y, w, h) in holes {
      for dy in 0..<h where y + dy < 14 { for dx in 0..<w where x + dx < 14 { g[y + dy][x + dx] = 0 } }
    }
    return g
  }
```

```swift
struct PixelSprite: View {
  let grid: PetSprites.Grid
  let pixelSize: CGFloat           // e.g. 1 for menubar strip, 2 for dropdown
  let amber: Color
  let ivory: Color
  let shadow: Color

       amber: Color = Dial.Palette.amber,
       ivory: Color = Dial.Palette.ivory,
       shadow: Color = Dial.Palette.muted) {

    self.amber = amber; self.ivory = ivory; self.shadow = shadow

  var body: some View {
    Canvas { ctx, size in
      let w = size.width / CGFloat(grid[0].count)
      let h = size.height / CGFloat(grid.count)
      for (y, row) in grid.enumerated() {
        for (x, v) in row.enumerated() where v != 0 {
          let rect = CGRect(x: CGFloat(x) * w, y: CGFloat(y) * h, width: w, height: h)
          let color = (v == 1 ? amber : v == 2 ? ivory : shadow)
          ctx.fill(Path(rect), with: .color(color))
        }
      }
    }
    .frame(width: CGFloat(grid[0].count) * pixelSize,
           height: CGFloat(grid.count) * pixelSize)
  }
}
```

```swift
@Observable final class AppCoordinator {
  let dbq: DatabaseQueue
  let client: SpotifyClient
  let poller: PollingService
  let engine: PetEngine
  let aggCache: AggregationsCache
  let nowPlaying: NowPlayingViewModel
  let menubar: MenubarStripViewModel
  let pet: PetViewModel
  let settings: SettingsViewModel

  init(dbq: DatabaseQueue, client: SpotifyClient, auth: SpotifyAuthService) {
    self.dbq = dbq
    self.client = client
    self.engine = PetEngine(dbq: dbq)
    self.aggCache = AggregationsCache(agg: PlayAggregations(dbq: dbq))
    self.poller = PollingService(client: client, interval: 3.0)
    self.nowPlaying = NowPlayingViewModel(agg: aggCache)
    self.menubar = MenubarStripViewModel()
    self.pet = PetViewModel()
    self.settings = SettingsViewModel(auth: auth, dbq: dbq)
  }

  func start() {
    Task {
      do {
        for try await event in poller.stream() { await handle(event) }
      } catch { print("[Coordinator] polling error: \(error)") }
    }
  }

  private func handle(_ event: PlaybackEvent) async {
    switch event {
    case .started(let uri, let name, let artistName, let artistUri, let albumName, let artUrl, _, _):
      nowPlaying.track = .init(uri: uri, name: name, artistName: artistName, artistUri: artistUri,
                               albumName: albumName, artUrl: artUrl)
      menubar.track = nowPlaying.track
      menubar.state = .playing
      aggCache.refresh(trackUri: uri, artistUri: artistUri)
      nowPlaying.plays30d = aggCache.plays30d
      // Fetch BPM for beat-sync (ignore errors)
      if let id = uri.split(separator: ":").last {
        if let f = try? await client.audioFeatures(trackId: String(id)) {
          pet.currentBPM = f.tempo
        }
      }
      resolvePetBehavior()
    case .paused:
      menubar.state = .paused; resolvePetBehavior()
    case .resumed:
      menubar.state = .playing; resolvePetBehavior()
    case .noPlayback:
      menubar.state = .nothingPlaying; resolvePetBehavior()
    case .countedPlay(let uri, let ms, let at):
      try? engine.recordPlay(.init(playedAt: at, trackUri: uri, msPlayed: ms, source: "live"))
      aggCache.refresh(trackUri: uri, artistUri: nowPlaying.track?.artistUri)
      nowPlaying.plays30d = aggCache.plays30d
      pet.syncFromStore(dbq: dbq)
      resolvePetBehavior()
    case .progress(let ms):
      menubar.progressMs = ms
    }
  }

  private func resolvePetBehavior() {
    let ctx = PetContext(
      playing: menubar.state == .playing,
      paused:  menubar.state == .paused,
      firstListen: aggCache.firstListen,
      rediscovered: aggCache.rediscovered,
      leveledUp: pet.justLeveledUp,
      lastPlayAt: pet.lastPlayAt,
      now: Int64(Date().timeIntervalSince1970),
      bpm: pet.currentBPM
    )
    pet.behavior = PetBehaviorResolver.resolve(ctx)
  }
}
```

```swift
      let wasStage = stage

      xp = s.xp
```

```swift
// Dial/History/Importer.swift
import GRDB
import Foundation

final class Importer {
  private let dbq: DatabaseQueue
  private let engine: PetEngine
  init(dbq: DatabaseQueue, engine: PetEngine) { self.dbq = dbq; self.engine = engine }

  @discardableResult
  func importBatch(_ rows: [EndsongRow], sourcePath: String) throws -> Int {
    var inserted = 0
    try dbq.write { db in
      for r in rows {
        // upsert a minimal Track so aggregations (e.g. artistRank7d) have metadata
        let artistsJson = r.artistName.map { #"[{"name":"\#($0)","uri":"artist:\#($0.replacingOccurrences(of: " ", with: "_"))"}]"# }
        try Track(uri: r.trackUri, name: r.trackName, albumName: r.albumName,
                  albumArtUrl: nil, durationMs: nil, artistsJson: artistsJson).save(db)
        // insert play; unique constraint rejects dupes
        do {
          var p = Play(playedAt: r.playedAt, trackUri: r.trackUri, msPlayed: r.msPlayed, source: "import")
          try p.insert(db)
          inserted += 1
          // award XP via engine (re-fetched to keep transaction single)
          try applyXPInTxn(for: p, in: db)
        } catch DatabaseError.SQLITE_CONSTRAINT { /* duplicate — skip */ }
      }
      var e = ImportLogEntry(importedAt: Int64(Date().timeIntervalSince1970),
                             sourcePath: sourcePath, rowsInserted: Int64(inserted), notes: nil)
      try e.insert(db)
    }
    return inserted
  }

  private func applyXPInTxn(for play: Play, in db: Database) throws {
    var state = try PetState.fetchOne(db, key: 1)!
    let streakDay = StreakCalendar.ymd(from: play.playedAt)
    let streak = StreakCalendar().apply(playDate: streakDay, priorAnchor: state.streakAnchorDate, priorDays: state.streakDays)
    let xpDelta = XPFormula.compute(msPlayed: play.msPlayed, variety: .none, streakDays: streak.days)
    state.xp = min(state.xp + xpDelta, 25_000)   // import cap
    state.streakAnchorDate = streak.anchor; state.streakDays = streak.days
    state.lastPlayAt = play.playedAt
    state.stage = PetStageTable.stage(forXP: state.xp, elderUnlocked: false)
```

```swift
// Dial/History/MetadataEnricher.swift
import GRDB
import Foundation

final class MetadataEnricher {
  private let dbq: DatabaseQueue
  private let client: SpotifyClient
  init(dbq: DatabaseQueue, client: SpotifyClient) { self.dbq = dbq; self.client = client }

  func enrichMissing(batchSize: Int = 50, rateLimitPerSec: Double = 5) async throws {
    let missing: [String] = try dbq.read { db in
      try String.fetchAll(db, sql: """
        SELECT t.uri FROM tracks t WHERE t.album_art_url IS NULL
      """)
    }
    for chunk in missing.chunked(into: batchSize) {
      let metas = try await client.tracks(uris: chunk)
      try dbq.write { db in
        for m in metas {
          try Track(uri: m.uri, name: m.name, albumName: m.album.name,
                    albumArtUrl: m.album.images.first?.url, durationMs: Int64(m.duration_ms),
                    artistsJson: try encodeArtists(m.artists)).save(db)
        }
      }
      try await Task.sleep(nanoseconds: UInt64(1e9 / rateLimitPerSec))
    }
  }
  private func encodeArtists(_ a: [TrackMeta.Artist]) throws -> String {
    struct E: Encodable { let name: String; let uri: String }
    let data = try JSONEncoder().encode(a.map { E(name: $0.name, uri: $0.uri) })
    return String(data: data, encoding: .utf8) ?? "[]"
  }
}
private extension Array {
  func chunked(into size: Int) -> [[Element]] {
    stride(from: 0, to: count, by: size).map { Array(self[$0..<Swift.min($0+size, count)]) }
  }
}
```
